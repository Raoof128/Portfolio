import {chromium} from '/workspace/Portfolio/node_modules/playwright/index.mjs';
import {readFile,writeFile} from 'node:fs/promises';
const browser=await chromium.launch({executablePath:'/usr/bin/chromium'});const result={buildId:(await readFile('.next/BUILD_ID','utf8')).trim(),version:JSON.parse(await readFile('out/version.json','utf8')),cases:[]};
const cases=[['en','light',1440,false],['en','dark',1440,false],['fa','dark',320,false],['ar','light',390,false],['zh','light',390,false],['es','light',320,false],['en','dark',1440,true]];
for(const [locale,theme,width,forced] of cases){const context=await browser.newContext({viewport:{width,height:900},reducedMotion:'reduce',colorScheme:theme,forcedColors:forced?'active':'none',permissions:['clipboard-read','clipboard-write']});await context.addInitScript(theme=>localStorage.setItem('researchos.appearance.v1',theme),theme);const p=await context.newPage();const errors=[];p.on('pageerror',e=>errors.push(e.message));await p.goto('http://127.0.0.1:4398'+(locale==='en'?'':'/'+locale)+'/projects/project-simurgh',{waitUntil:'networkidle'});const terminal=p.locator('.os-terminal').first();await terminal.scrollIntoViewIfNeeded();const button=terminal.locator('.os-terminal-copy');const slug=`terminal-${locale}-${theme}-${width}${forced?'-forced':''}`;const metrics=async()=>terminal.evaluate(e=>{const t=e.querySelector('.title-bar'),c=e.querySelector('.window-body'),b=e.querySelector('button'),s=getComputedStyle(e);return {frame:e.getBoundingClientRect().toJSON(),header:t.getBoundingClientRect().toJSON(),body:c.getBoundingClientRect().toJSON(),button:b.getBoundingClientRect().toJSON(),frameRadius:s.borderRadius,frameShadow:s.boxShadow,frameGlass:getComputedStyle(e,'::before').opacity,frameBlur:getComputedStyle(e,'::after').backdropFilter,titleColour:getComputedStyle(t).color,consoleColour:getComputedStyle(c).color,consoleBackground:getComputedStyle(c).backgroundColor,buttonFont:getComputedStyle(b).font,buttonBackground:getComputedStyle(b).backgroundImage,buttonBefore:getComputedStyle(b,'::before').opacity,buttonAfter:getComputedStyle(b,'::after').opacity,pageOverflow:document.documentElement.scrollWidth>innerWidth,terminalOverflow:e.scrollWidth>e.clientWidth+1}});const states={default:await metrics()};await terminal.screenshot({path:`/tmp/aero-round2-accepted-windows/${slug}-default.png`});if(locale==='en'&&width===1440&&!forced){await button.hover();states.hover=await metrics();await terminal.screenshot({path:`/tmp/aero-round2-accepted-windows/${slug}-hover.png`});const box=await button.boundingBox();await p.mouse.move(box.x+box.width/2,box.y+box.height/2);await p.mouse.down();states.pressed=await metrics();await terminal.screenshot({path:`/tmp/aero-round2-accepted-windows/${slug}-pressed.png`});await p.mouse.up();await p.mouse.move(0,0);await p.keyboard.press('Tab');await button.focus();states.focus=await metrics();await terminal.screenshot({path:`/tmp/aero-round2-accepted-windows/${slug}-focus.png`});}
const expected=await terminal.locator('code').allTextContents();await button.click();await p.waitForTimeout(50);const clipboard=await p.evaluate(()=>navigator.clipboard.readText());const copied={matches:clipboard===expected.join('\n'),button:await button.textContent(),status:await terminal.getByRole('status').textContent()};result.cases.push({locale,theme,width,forced,errors,states,copied});await context.close();}
await browser.close();await writeFile('/tmp/aero-round2-accepted-windows/terminal-compiled.json',JSON.stringify(result,null,2));console.log(JSON.stringify({build:result.buildId,cases:result.cases.map(r=>({locale:r.locale,theme:r.theme,width:r.width,forced:r.forced,errors:r.errors,frameHeight:r.states.default.frame.height,headerHeight:r.states.default.header.height,buttonHeight:r.states.default.button.height,pageOverflow:r.states.default.pageOverflow,terminalOverflow:r.states.default.terminalOverflow,copied:r.copied}))},null,2));
