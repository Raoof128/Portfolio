import {chromium} from '/workspace/Portfolio/node_modules/playwright/index.mjs';
import {readFile,writeFile} from 'node:fs/promises';
const b=await chromium.launch({executablePath:'/usr/bin/chromium'});const p=await b.newPage({viewport:{width:1440,height:900},reducedMotion:'reduce'});await p.goto('http://127.0.0.1:4398/projects',{waitUntil:'networkidle'});const grip=p.locator('section.os-workspace-window>.os-window-resize');await grip.hover();await p.waitForTimeout(600);await p.screenshot({path:'/tmp/aero-round2-accepted-windows/resize-hover-compiled.png'});const hover=await grip.evaluate(e=>{const c=getComputedStyle(e,'::after'),r=e.getBoundingClientRect();return {rect:r.toJSON(),clip:getComputedStyle(e).clipPath,tooltip:{content:c.content,width:c.width,height:c.height,top:c.top,right:c.right,bottom:c.bottom,left:c.left,opacity:c.opacity},overTooltip:document.elementFromPoint(r.right-40,r.top-15)?.closest('button')===e}});await p.mouse.move(0,0);await grip.focus();await p.waitForTimeout(600);await p.screenshot({path:'/tmp/aero-round2-accepted-windows/resize-focus-compiled.png'});const focus=await grip.evaluate(e=>({focused:e===document.activeElement,outline:getComputedStyle(e).outline,tooltip:getComputedStyle(e,'::after').opacity}));await p.keyboard.press('Escape');const escape=await grip.evaluate(e=>({dismissed:e.dataset.tooltipDismissed,tooltip:getComputedStyle(e,'::after').opacity}));await p.locator('section.os-workspace-window .os-window-controls>summary').click();const footer=await p.locator('.os-window-controls-footer').evaluate(e=>({columns:getComputedStyle(e).gridTemplateColumns,buttons:[...e.querySelectorAll('button')].map(b=>({label:b.textContent,rect:b.getBoundingClientRect().toJSON()}))}));await p.screenshot({path:'/tmp/aero-round2-accepted-windows/footer-compiled.png'});await writeFile('/tmp/aero-round2-accepted-windows/resize-compiled.json',JSON.stringify({proposalInjected:false,hover,focus,escape,footer},null,2));console.log(JSON.stringify({hover,focus,escape,footer},null,2));await b.close();
